```dart
import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:intl/intl.dart';

import '../../../core/providers/current_property_provider.dart';
import '../data/piquete_model.dart';
import '../data/piquete_repository.dart';

class PiqueteFormDialog extends ConsumerStatefulWidget {
  const PiqueteFormDialog({this.editing, super.key});
  final Piquete? editing;

  static Future<Piquete?> show(BuildContext context, {Piquete? editing}) {
    final isWide = MediaQuery.of(context).size.width >= 600;
    if (isWide) {
      return showDialog<Piquete?>(
        context: context,
        builder: (_) => Dialog(
          child: ConstrainedBox(
            constraints: const BoxConstraints(maxWidth: 480),
            child: PiqueteFormDialog(editing: editing),
          ),
        ),
      );
    }
    return showModalBottomSheet<Piquete?>(
      context: context,
      isScrollControlled: true,
      builder: (ctx) => Padding(
        padding: EdgeInsets.only(
          bottom: MediaQuery.of(ctx).viewInsets.bottom,
        ),
        child: PiqueteFormDialog(editing: editing),
      ),
    );
  }

  @override
  ConsumerState<PiqueteFormDialog> createState() => _PiqueteFormDialogState();
}

class _PiqueteFormDialogState extends ConsumerState<PiqueteFormDialog> {
  final _formKey = GlobalKey<FormState>();
  late final TextEditingController _nomeCtrl;
  late final TextEditingController _areaCtrl;
  late final TextEditingController _capCtrl;
  bool _busy = false;

  static final _ptBr = NumberFormat.decimalPattern('pt_BR');

  @override
  void initState() {
    super.initState();
    _nomeCtrl = TextEditingController(text: widget.editing?.nome ?? '');
    _areaCtrl = TextEditingController(
      text: widget.editing != null ? _ptBr.format(widget.editing!.areaHa) : '',
    );
    _capCtrl = TextEditingController(
      text: widget.editing != null
          ? _ptBr.format(widget.editing!.capacidadeUa)
          : '',
    );
  }

  @override
  void dispose() {
    _nomeCtrl.dispose();
    _areaCtrl.dispose();
    _capCtrl.dispose();
    super.dispose();
  }

  String? _required(String? v) =>
      (v == null || v.trim().isEmpty) ? 'Campo obrigatório' : null;

  /// Validator + parser for pt-BR comma decimals (Pitfall 7).
  String? _validateDecimal(String? v) {
    if (v == null || v.trim().isEmpty) return 'Campo obrigatório';
    final normalized = v.trim().replaceAll('.', '').replaceAll(',', '.');
    final parsed = double.tryParse(normalized);
    if (parsed == null || parsed <= 0) return 'Valor deve ser maior que zero';
    return null;
  }

  double _parseDecimal(String v) =>
      double.parse(v.trim().replaceAll('.', '').replaceAll(',', '.'));

  Future<void> _submit() async {
    if (!_formKey.currentState!.validate()) return;
    final property = ref.read(currentPropertyProvider).asData?.value;
    if (property == null) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Selecione uma propriedade primeiro.')),
      );
      return;
    }
    setState(() => _busy = true);
    try {
      final repo = ref.read(piqueteRepositoryProvider);
      final Piquete result;
      if (widget.editing != null) {
        result = await repo.updatePiquete(
          id: widget.editing!.id,
          nome: _nomeCtrl.text.trim(),
          areaHa: _parseDecimal(_areaCtrl.text),
          capacidadeUa: _parseDecimal(_capCtrl.text),
        );
      } else {
        result = await repo.createPiquete(
          propriedadeId: property.id,
          nome: _nomeCtrl.text.trim(),
          areaHa: _parseDecimal(_areaCtrl.text),
          capacidadeUa: _parseDecimal(_capCtrl.text),
        );
      }
      ref.invalidate(piqueteListProvider);
      if (!mounted) return;
      Navigator.of(context).pop(result);
    } catch (_) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text(
            'Não foi possível salvar. Verifique sua conexão e tente novamente.',
          ),
        ),
      );
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);
    final isEdit = widget.editing != null;
    return Padding(
      padding: const EdgeInsets.all(24),
      child: Form(
        key: _formKey,
        autovalidateMode: AutovalidateMode.onUserInteraction,
        child: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            Text(
              isEdit ? 'Editar piquete' : 'Criar piquete',
              style: theme.textTheme.titleLarge
                  ?.copyWith(fontWeight: FontWeight.w600),
            ),
            const SizedBox(height: 24),
            TextFormField(
              controller: _nomeCtrl,
              decoration: const InputDecoration(
                labelText: 'Nome',
                border: OutlineInputBorder(),
              ),
              validator: _required,
            ),
            const SizedBox(height: 16),
            TextFormField(
              controller: _areaCtrl,
              keyboardType: const TextInputType.numberWithOptions(
                decimal: true,
              ),
              inputFormatters: [
                FilteringTextInputFormatter.allow(RegExp(r'[0-9.,]')),
              ],
              decoration: const InputDecoration(
                labelText: 'Área (ha)',
                border: OutlineInputBorder(),
              ),
              validator: _validateDecimal,
            ),
            const SizedBox(height: 16),
            TextFormField(
              controller: _capCtrl,
              keyboardType: const TextInputType.numberWithOptions(
                decimal: true,
              ),
              inputFormatters: [
                FilteringTextInputFormatter.allow(RegExp(r'[0-9.,]')),
              ],
              decoration: const InputDecoration(
                labelText: 'Capacidade (UA)',
                border: OutlineInputBorder(),
              ),
              validator: _validateDecimal,
            ),
            const SizedBox(height: 24),
            Row(
              mainAxisAlignment: MainAxisAlignment.end,
              children: [
                TextButton(
                  onPressed: _busy ? null : () => Navigator.of(context).pop(),
                  child: const Text('Cancelar'),
                ),
                const SizedBox(width: 8),
                FilledButton(
                  onPressed: _busy ? null : _submit,
                  style: FilledButton.styleFrom(
                    minimumSize: const Size(120, 48),
                  ),
                  child: _busy
                      ? const SizedBox(
                          width: 20,
                          height: 20,
                          child: CircularProgressIndicator(strokeWidth: 2),
                        )
                      : const Text('Salvar'),
                ),
              ],
            ),
          ],
        ),
      ),
    );
  }
}
```

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:go_router/go_router.dart';
import 'package:intl/intl.dart';

import '../../../core/providers/current_property_provider.dart';
import '../../propriedades/providers/perfil_provider.dart';
import '../data/piquete_model.dart';
import '../data/piquete_repository.dart';
import 'piquete_form_dialog.dart';

class PiqueteDetailScreen extends ConsumerWidget {
  const PiqueteDetailScreen({required this.id, super.key});
  final String id;

  static final _ptBr = NumberFormat.decimalPattern('pt_BR');

  Future<void> _confirmDelete(
    BuildContext context,
    WidgetRef ref,
    Piquete p,
  ) async {
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('Excluir piquete?'),
        content: Text(
          'O piquete "${p.nome}" será removido. Esta ação não pode ser desfeita.',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.of(ctx).pop(false),
            child: const Text('Cancelar'),
          ),
          FilledButton(
            style: FilledButton.styleFrom(
              backgroundColor: Theme.of(ctx).colorScheme.error,
            ),
            onPressed: () => Navigator.of(ctx).pop(true),
            child: const Text('Excluir'),
          ),
        ],
      ),
    );
    if (confirmed != true) return;
    try {
      await ref.read(piqueteRepositoryProvider).softDeletePiquete(p.id);
      ref.invalidate(piqueteListProvider);
      if (!context.mounted) return;
      context.pop();
    } catch (_) {
      if (!context.mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text(
            'Não foi possível salvar. Verifique sua conexão e tente novamente.',
          ),
        ),
      );
    }
  }

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final piqueteAsync = ref.watch(piqueteByIdProvider(id));
    final perfilAsync = ref.watch(perfilForCurrentPropertyProvider);
    final activeProp = ref.watch(currentPropertyProvider).asData?.value;
    final isVet = perfilAsync.maybeWhen(
      data: (p) => p == 'veterinario',
      orElse: () => false,
    );
    final theme = Theme.of(context);

    return Scaffold(
      appBar: AppBar(
        title: Text(piqueteAsync.asData?.value?.nome ?? 'Piquete'),
        actions: !isVet || piqueteAsync.asData?.value == null
            ? const []
            : [
                IconButton(
                  icon: const Icon(Icons.edit_outlined),
                  tooltip: 'Editar piquete',
                  onPressed: () async {
                    final updated = await PiqueteFormDialog.show(
                      context,
                      editing: piqueteAsync.asData!.value,
                    );
                    if (updated != null) {
                      ref.invalidate(piqueteByIdProvider(id));
                    }
                  },
                ),
                IconButton(
                  icon: const Icon(Icons.delete_outline),
                  tooltip: 'Excluir piquete',
                  onPressed: () =>
                      _confirmDelete(context, ref, piqueteAsync.asData!.value!),
                ),
              ],
      ),
      body: piqueteAsync.when(
        data: (piquete) {
          if (piquete == null) {
            return Center(
              child: Padding(
                padding: const EdgeInsets.all(24),
                child: Text(
                  'Piquete não encontrado.',
                  textAlign: TextAlign.center,
                  style: theme.textTheme.bodyMedium,
                ),
              ),
            );
```
