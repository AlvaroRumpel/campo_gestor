```dart
import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../../../core/providers/current_property_provider.dart';
import '../../auth/data/property_repository.dart';

/// Returns the perfil string ('veterinario' | 'proprietario' | 'leitor' | null)
/// for the active propriedade. Null when no active propriedade or user is not
/// a member.
///
/// Used by widgets to gate role-based visibility (D-06, D-07).
final perfilForCurrentPropertyProvider =
    FutureProvider<String?>((ref) async {
  final property = await ref.watch(currentPropertyProvider.future);
  if (property == null) return null;
  final memberships = await ref.watch(memberPropertiesProvider.future);
  for (final m in memberships) {
    if (m.property.id == property.id) return m.perfil;
  }
  return null;
});
```

Adaptive form widget — used inside both `showDialog` (wide ≥600dp) and `showModalBottomSheet` (narrow). Includes a static helper `PropertyFormDialog.show(...)` that picks the right presentation.

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../../auth/data/property_repository.dart';
import '../data/propriedade_model.dart';
import '../data/propriedade_repository.dart';

/// Form for create/edit of a Propriedade.
/// - Wide (>=600dp): wrapped in AlertDialog by `show()`.
/// - Narrow (<600dp): wrapped in ModalBottomSheet with viewInsets padding (Pitfall 5).
class PropertyFormDialog extends ConsumerStatefulWidget {
  const PropertyFormDialog({this.editing, super.key});
  final Propriedade? editing;

  /// Adaptive presenter. Returns the created/edited Propriedade, or null on cancel.
  static Future<Propriedade?> show(
    BuildContext context, {
    Propriedade? editing,
  }) {
    final isWide = MediaQuery.of(context).size.width >= 600;
    if (isWide) {
      return showDialog<Propriedade?>(
        context: context,
        builder: (_) => Dialog(
          child: ConstrainedBox(
            constraints: const BoxConstraints(maxWidth: 480),
            child: PropertyFormDialog(editing: editing),
          ),
        ),
      );
    }
    return showModalBottomSheet<Propriedade?>(
      context: context,
      isScrollControlled: true,
      builder: (ctx) => Padding(
        padding: EdgeInsets.only(
          bottom: MediaQuery.of(ctx).viewInsets.bottom,
        ),
        child: PropertyFormDialog(editing: editing),
      ),
    );
  }

  @override
  ConsumerState<PropertyFormDialog> createState() => _PropertyFormDialogState();
}

class _PropertyFormDialogState extends ConsumerState<PropertyFormDialog> {
  final _formKey = GlobalKey<FormState>();
  late final TextEditingController _nomeCtrl;
  late final TextEditingController _proprietarioCtrl;
  bool _busy = false;

  @override
  void initState() {
    super.initState();
    _nomeCtrl = TextEditingController(text: widget.editing?.nome ?? '');
    _proprietarioCtrl =
        TextEditingController(text: widget.editing?.proprietario ?? '');
  }

  @override
  void dispose() {
    _nomeCtrl.dispose();
    _proprietarioCtrl.dispose();
    super.dispose();
  }

  String? _required(String? v) =>
      (v == null || v.trim().isEmpty) ? 'Campo obrigatório' : null;

  Future<void> _submit() async {
    if (!_formKey.currentState!.validate()) return;
    setState(() => _busy = true);
    try {
      final repo = ref.read(propriedadeRepositoryProvider);
      final Propriedade result;
      if (widget.editing != null) {
        result = await repo.updatePropriedade(
          id: widget.editing!.id,
          nome: _nomeCtrl.text.trim(),
          proprietario: _proprietarioCtrl.text.trim(),
        );
      } else {
        result = await repo.createPropriedadeWithMembership(
          nome: _nomeCtrl.text.trim(),
          proprietario: _proprietarioCtrl.text.trim(),
        );
      }
      // Pitfall 6: invalidate so UI refreshes. memberPropertiesProvider feeds
      // PropertySelector — must be invalidated when a new propriedade exists.
      ref.invalidate(propriedadeListProvider);
      ref.invalidate(memberPropertiesProvider);
      if (!mounted) return;
      Navigator.of(context).pop(result);
    } catch (_) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text(
            'Não foi possível salvar. Verifique sua conexão e tente novamente.',
          ),
        ),
      );
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);
    final isEdit = widget.editing != null;
    return Padding(
      padding: const EdgeInsets.all(24),
      child: Form(
        key: _formKey,
        autovalidateMode: AutovalidateMode.onUserInteraction,
        child: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            Text(
              isEdit ? 'Editar fazenda' : 'Criar fazenda',
              style: theme.textTheme.titleLarge
                  ?.copyWith(fontWeight: FontWeight.w600),
            ),
            const SizedBox(height: 24),
            TextFormField(
              controller: _nomeCtrl,
              decoration: const InputDecoration(
                labelText: 'Nome da fazenda',
                border: OutlineInputBorder(),
              ),
              validator: _required,
            ),
            const SizedBox(height: 16),
            TextFormField(
              controller: _proprietarioCtrl,
              decoration: const InputDecoration(
                labelText: 'Proprietário',
                border: OutlineInputBorder(),
              ),
              validator: _required,
            ),
            const SizedBox(height: 24),
            Row(
              mainAxisAlignment: MainAxisAlignment.end,
              children: [
                TextButton(
                  onPressed: _busy ? null : () => Navigator.of(context).pop(),
                  child: const Text('Cancelar'),
                ),
                const SizedBox(width: 8),
                FilledButton(
                  onPressed: _busy ? null : _submit,
                  style: FilledButton.styleFrom(
                    minimumSize: const Size(120, 48),
                  ),
                  child: _busy
                      ? const SizedBox(
                          width: 20,
                          height: 20,
                          child: CircularProgressIndicator(strokeWidth: 2),
                        )
                      : const Text('Salvar'),
                ),
              ],
            ),
          ],
        ),
      ),
    );
  }
}
```

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../../../core/widgets/empty_state.dart';
import '../../auth/data/property_repository.dart';
import '../data/propriedade_model.dart';
import '../data/propriedade_repository.dart';
import '../providers/perfil_provider.dart';
import 'property_form_dialog.dart';

class PropriedadesScreen extends ConsumerWidget {
  const PropriedadesScreen({super.key});

  Future<void> _confirmDelete(
    BuildContext context,
    WidgetRef ref,
    Propriedade p,
  ) async {
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('Excluir fazenda?'),
        content: Text(
          'A fazenda "${p.nome}" será removida. Esta ação não pode ser desfeita.',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.of(ctx).pop(false),
            child: const Text('Cancelar'),
          ),
          FilledButton(
            style: FilledButton.styleFrom(
              backgroundColor: Theme.of(ctx).colorScheme.error,
            ),
            onPressed: () => Navigator.of(ctx).pop(true),
            child: const Text('Excluir'),
          ),
        ],
      ),
    );
    if (confirmed != true) return;
    try {
      await ref.read(propriedadeRepositoryProvider).softDeletePropriedade(p.id);
      ref.invalidate(propriedadeListProvider);
      ref.invalidate(memberPropertiesProvider);
    } catch (_) {
      if (!context.mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text(
            'Não foi possível salvar. Verifique sua conexão e tente novamente.',
          ),
        ),
      );
    }
  }

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final list = ref.watch(propriedadeListProvider);
    // Show FAB / edit / delete only for any veterinario membership the user
    // has — propriedades screen is cross-property, so we treat the user as
    // "able to manage" if they are veterinario in at least one property.
    final memberships = ref.watch(memberPropertiesProvider);
    final canManage = memberships.maybeWhen(
      data: (m) => m.any((x) => x.perfil == 'veterinario'),
      orElse: () => false,
    );
    // Allow create for any authenticated user (D-04 — first-time creator path).
    final canCreate = true;

    return Scaffold(
      appBar: AppBar(title: const Text('Fazendas')),
      body: list.when(
        data: (items) {
          if (items.isEmpty) {
            return const EmptyState(
              icon: Icons.home_work_outlined,
              heading: 'Nenhuma fazenda cadastrada',
              body:
                  'Crie sua primeira fazenda para começar a organizar o rebanho.',
            );
          }
          return ListView.separated(
            itemCount: items.length,
            separatorBuilder: (_, __) => const Divider(height: 1),
            itemBuilder: (ctx, i) {
              final p = items[i];
              return ListTile(
                leading: const Icon(Icons.home_work_outlined),
                title: Text(p.nome),
                subtitle: Text(p.proprietario),
                trailing: canManage
                    ? PopupMenuButton<String>(
                        tooltip: 'Opções',
                        onSelected: (value) async {
                          if (value == 'edit') {
                            await PropertyFormDialog.show(context, editing: p);
                          } else if (value == 'delete') {
                            await _confirmDelete(context, ref, p);
                          }
                        },
                        itemBuilder: (_) => const [
                          PopupMenuItem(value: 'edit', child: Text('Editar')),
                          PopupMenuItem(value: 'delete', child: Text('Excluir')),
                        ],
                      )
                    : null,
              );
            },
          );
        },
        loading: () => const Center(child: CircularProgressIndicator()),
        error: (_, __) => Center(
          child: Padding(
            padding: const EdgeInsets.all(24),
            child: Text(
              'Não foi possível carregar os dados. Verifique sua conexão e tente novamente.',
              textAlign: TextAlign.center,
              style: Theme.of(context).textTheme.bodyMedium,
            ),
          ),
        ),
      ),
      floatingActionButton: canCreate
          ? FloatingActionButton(
              tooltip: 'Criar fazenda',
              onPressed: () => PropertyFormDialog.show(context),
              child: const Icon(Icons.add),
```
